title_is_grand_patriarch = {
    highest_held_title_tier = tier_empire
    has_government = clan_government
    any_held_title = { is_head_of_faith = no }
}

title_is_cardinal = {
    has_character_flag = cardinal_flag
    exists = clerical_elector_title
    primary_title ?= clerical_elector_title
    pam_is_antipope_trigger = no
    NOT = { religious_head_or_challenger ?= this }
    NOT = { has_trait = saint }
}

title_is_confessor = {
    has_council_position = councillor_court_chaplain
    is_ruler = no
    court_owner ?= {
        religion = religion:christianity_religion
        government_has_flag = government_is_theocracy
    }
}

title_is_archbishop = {
    has_government = ecclesiastical_government
    title_is_confessor = no

    OR = {
        faith = faith:catholic
        rite = {
            rite_has_doctrine = special_doctrine_is_western_christian_faith
        }
    }

    title_is_cardinal = no

    pam_is_antipope_trigger = no
    NOT = { religious_head_or_challenger ?= this }
    NOT = { has_trait = saint }

    OR = {
        # Titulaire personnel d’une région cléricale.
        any_held_title = {
            tier = tier_duchy
            has_clerical_region = yes
        }

        # Chapelain de la cour d’un archevêché.
        AND = {
            has_council_position = councillor_court_chaplain
            liege.primary_title ?= {
                tier = tier_duchy
                has_clerical_region = yes
            }
        }
    }

    # Aucun pouvoir territorial donnant un préfixe.
    title_is_prince_or_duke_archbishop = no
}

# Évèque / Évèchesse
title_is_bishop = {
    has_government = ecclesiastical_government
    title_is_confessor = no

    OR = {
        has_council_position = councillor_court_chaplain
        highest_held_title_tier = tier_barony
    }

    OR = {
        faith = faith:catholic
        rite = {
            rite_has_doctrine = special_doctrine_is_western_christian_faith
        }
    }

    pam_is_antipope_trigger = no
    NOT = { religious_head_or_challenger ?= this }
    NOT = { exists = clerical_elector_title }
    NOT = { has_trait = saint }

    # Sinon GetActualBishopTitle produit archevêque ou davantage.
    NOT = {
        any_held_title = {
            tier = tier_duchy
            has_clerical_region = yes
        }
    }
    NOT = {
        liege.primary_title ?= {
            tier = tier_duchy
            has_clerical_region = yes
        }
    }
}

title_is_prince_or_duke_archbishop = {
    has_government = ecclesiastical_government

    OR = {
        faith = faith:catholic
        rite = {
            rite_has_doctrine = special_doctrine_is_western_christian_faith
        }
    }

    pam_is_antipope_trigger = no
    NOT = { religious_head_or_challenger ?= this }
    NOT = { exists = clerical_elector_title }
    NOT = { has_trait = saint }

    # Charge d’archevêque.
    any_held_title = {
        tier = tier_duchy
        has_clerical_region = yes
    }

    # Pouvoir territorial donnant le préfixe.
    OR = {
        any_held_title = {
            tier = tier_county
            is_landless_type_title = no
        }
        any_held_title = {
            tier = tier_duchy
            is_landless_type_title = no
        }
    }
}

# "Empereur" par défaut, sans falvor (Basileus, Kaiser, etc)
title_is_emperor_default = {
    highest_held_title_tier = tier_empire
    NOR = {
        title_is_grand_patriarch = yes
        title_is_samrat = yes
        title_is_hwangje = yes
        title_is_caliph = yes
        has_character_flag = devaraja_flag
        has_title = title:k_chrysanthemum_throne
        culture = { has_cultural_pillar = heritage_central_germanic } # Kaiser
        culture = { has_name_list = name_list_greek } # Basileus
        AND = { # emperor_feudal_male_altaic_group (Khagan)
            OR = {
                has_government = feudal_government
                has_government = clan_government
                has_government = tribal_government
            }
            OR = {
                culture = { has_cultural_pillar = heritage_mongolic }
                culture = { has_cultural_pillar = heritage_turkic }
            }
        }
        has_government = tribal_government # emperor_tribal_male (Grand roi), emperor_tribal_female (Grande reine)
    }
}

title_is_caliph = {
    highest_held_title_tier = tier_empire
    OR = {
        AND = {
            highest_held_title_tier >= tier_kingdom
            OR = {
                has_government = theocracy_government
                has_government = ecclesiastical_government
                has_government = clan_government
            }
            religion = religion:islam_religion
        }
        OR = {
            has_title = title:d_sunni
            has_title = title:d_shiite
            has_title = title:h_dar_al_islam
        }
    }
}

title_is_emir = {
    is_female = no
    highest_held_title_tier = tier_duchy
    
    holds_landed_title = yes
    NOT = { joined_faction.faction_leader = this }
    title_has_khorezmshah_flavor = no
    title_has_independent_iranian_duke_flavor = no

    OR = {
        # Duc musulman clanique ou administratif, toute culture.
        AND = {
            religion = religion:islam_religion
            OR = {
                has_government = clan_government
                has_government = administrative_government
            }
        }

        # Exceptions culturelles.
        AND = {
            culture = { has_cultural_pillar = heritage_arabic }
            OR = {
                has_government = feudal_government
                has_government = clan_government
                has_government = administrative_government
            }
        }
        AND = {
            culture = { has_cultural_pillar = heritage_berber }
            OR = {
                has_government = feudal_government
                has_government = clan_government
                has_government = tribal_government
                has_government = administrative_government
            }
        }
        AND = {
            culture = { has_name_list = name_list_beja }
            OR = {
                has_government = feudal_government
                has_government = clan_government
                has_government = tribal_government
                has_government = administrative_government
                has_government = republic_government
            }
        }
        AND = {
            culture = { has_cultural_pillar = heritage_iberian }
            has_government = clan_government
        }
    }
}
