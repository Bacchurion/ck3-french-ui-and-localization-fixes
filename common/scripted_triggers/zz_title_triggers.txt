title_is_archbishop = {
    pam_is_antipope_trigger = no
    holds_landed_title = no
    OR = {
        faith = faith:catholic
        rite = {
            rite_has_doctrine = special_doctrine_is_western_christian_faith
        }
    }
    pam_is_archbishop_or_higher_trigger = yes
    has_government = ecclesiastical_government
    NOT = { has_trait = saint }
}

# "Empereur" par défaut, sans falvor (Basileus, Kaiser, etc)
title_is_emperor_default = {
    highest_held_title_tier = tier_empire
    NOR = {
        title_is_grand_patriarch = yes
        title_is_samrat = yes
        title_is_hwangje = yes
        has_character_flag = devaraja_flag
        has_title = title:k_chrysanthemum_throne
        culture = { has_cultural_pillar = heritage_central_germanic } # Kaiser
        culture = { has_name_list = name_list_greek } # Basileus
        AND = { # emperor_feudal_male_altaic_group (Khagan)
            OR = {
                has_government = feudal_government
                has_government = clan_government
                has_government = tribal_government
            }
            OR = {
                culture = { has_cultural_pillar = heritage_mongolic }
                culture = { has_cultural_pillar = heritage_turkic }
            }
        }
        has_government = tribal_government # emperor_tribal_male (Grand roi), emperor_tribal_female (Grande reine)
    }
}

title_is_emir = {
    is_female = no
    highest_held_title_tier = tier_duchy
    holds_landed_title = yes
    NOT = { joined_faction.faction_leader = this }

    OR = {
        # Duc musulman clanique ou administratif, toute culture.
        AND = {
            religion = religion:islam_religion
            OR = {
                has_government = clan_government
                has_government = administrative_government
            }
        }

        # Exceptions culturelles.
        AND = {
            culture = { has_cultural_pillar = heritage_arabic }
            OR = {
                has_government = feudal_government
                has_government = clan_government
                has_government = administrative_government
            }
        }
        AND = {
            culture = { has_cultural_pillar = heritage_berber }
            OR = {
                has_government = feudal_government
                has_government = clan_government
                has_government = tribal_government
                has_government = administrative_government
            }
        }
        AND = {
            culture = { has_name_list = name_list_beja }
            OR = {
                has_government = feudal_government
                has_government = clan_government
                has_government = tribal_government
                has_government = administrative_government
                has_government = republic_government
            }
        }
        AND = {
            culture = { has_cultural_pillar = heritage_iberian }
            has_government = clan_government
        }
    }
}
